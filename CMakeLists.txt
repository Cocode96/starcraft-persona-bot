cmake_minimum_required(VERSION 3.20)
project(StarCraftPersonaBot LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(persona_config STATIC bot/PersonaConfig.cpp bot/PersonaConfig.h)
target_include_directories(persona_config PUBLIC bot)
source_group("PersonaConfig" FILES bot/PersonaConfig.cpp bot/PersonaConfig.h)

enable_testing()
add_executable(config_tests tests/config_tests.cpp)
target_link_libraries(config_tests PRIVATE persona_config)
add_test(NAME profile_config_validation COMMAND config_tests)

option(BUILD_BWAPI_BOT "Build the Windows x86 BWAPI DLL" ${WIN32})
if(BUILD_BWAPI_BOT)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "BWAPI 4.4.0 needs Windows x86. Configure with -A Win32.")
    endif()
    set(BWAPI_ROOT "${CMAKE_SOURCE_DIR}/tools/bwapi/Release_Binary" CACHE PATH "BWAPI 4.4.0 SDK")
    if(NOT EXISTS "${BWAPI_ROOT}/include/BWAPI.h")
        message(FATAL_ERROR "Run python scripts/bootstrap.py --bwapi first.")
    endif()
    file(GLOB BWAPI_SOURCES CONFIGURE_DEPENDS "${BWAPI_ROOT}/BWAPILIB/Source/*.cpp")
    add_library(bwapi STATIC ${BWAPI_SOURCES} "${BWAPI_ROOT}/BWAPILIB/UnitCommand.cpp")
    target_include_directories(bwapi SYSTEM PUBLIC "${BWAPI_ROOT}/include")
    target_compile_definitions(bwapi PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
    add_library(PersonaBot SHARED bot/PersonaBot.cpp bot/PersonaBot.h bot/Dll.cpp)
    target_link_libraries(PersonaBot PRIVATE persona_config bwapi)
    add_executable(module_smoke_tests tests/module_smoke_tests.cpp)
    target_link_libraries(module_smoke_tests PRIVATE bwapi)
    add_dependencies(module_smoke_tests PersonaBot)
    add_test(NAME bwapi_module_load COMMAND module_smoke_tests $<TARGET_FILE:PersonaBot>)
    source_group("PersonaBot" FILES bot/PersonaBot.cpp bot/PersonaBot.h)
    source_group("Module" FILES bot/Dll.cpp)
    if(MSVC)
        target_compile_options(PersonaBot PRIVATE /W4 /utf-8)
        target_compile_options(persona_config PRIVATE /W4 /utf-8)
    endif()
endif()
